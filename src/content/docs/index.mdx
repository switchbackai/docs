<Section id="quickstart" overline="Quickstart" title="Three commands to your first Receipt">
  <div slot="lead">
    Install the client, log in with GitHub, and connect the GitHub App. Uploads
    start when login ends. The Receipt appears on your next push.
  </div>
  <Route>
    <Stop n={1} title="Install">
      <p slot="text">From the Homebrew tap.</p>

      ```sh frame="none" wrap
      brew install switchbackai/tap/switchbak
      ```
    </Stop>
    <Stop n={2} title="Log in">
      <p slot="text">Sign in with GitHub. Setup watches each installed agent.</p>

      ```sh frame="none" wrap
      switchbak login
      ```
    </Stop>
    <Stop n={3} title="Connect GitHub">
      <p slot="text">Install the App on your account or organization.</p>

      ```sh frame="none" wrap
      switchbak connect
      ```
    </Stop>
    <Stop destination title="Push">
      <p slot="text">The next push to a pull request gets its Receipt.</p>
      <div class="pills">
        <Pill tone="success">🧾 Receipt</Pill>
        <Pill>check run</Pill>
      </div>
    </Stop>
  </Route>
</Section>
